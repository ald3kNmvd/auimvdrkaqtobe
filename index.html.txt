<script>

/* 
   Меняем название поля в зависимости
   от выбранного статуса
*/

const status = document.getElementById("status");

const institutionLabel =
    document.getElementById("institutionLabel");

const institution =
    document.getElementById("institution");


status.addEventListener("change", function() {

    if (this.value === "school") {

        institutionLabel.textContent = "Школа";

        institution.placeholder =
            "Введите название школы";

    }

    else if (this.value === "college") {

        institutionLabel.textContent = "Колледж";

        institution.placeholder =
            "Введите название колледжа";

    }

    else if (this.value === "university") {

        institutionLabel.textContent = "Университет";

        institution.placeholder =
            "Введите название университета";

    }

});


/*
   Отправка формы
*/

document
.getElementById("studentForm")
.addEventListener("submit", function(event) {

    event.preventDefault();

    const city =
        document.getElementById("city").value;

    const institutionValue =
        document.getElementById("institution").value;

    const statusValue =
        document.getElementById("status").value;

    const direction =
        document.getElementById("direction").value;


    /*
       Сохраняем данные пока в браузере.
    */

    localStorage.setItem(
        "city",
        city
    );

    localStorage.setItem(
        "institution",
        institutionValue
    );

    localStorage.setItem(
        "status",
        statusValue
    );

    localStorage.setItem(
        "direction",
        direction
    );


    alert(
        "Спасибо! Форма заполнена."
    );


    /*
       Позже здесь поставим ссылку
       на сайт АИЮ МВД РК.

       Например:

       window.location.href =
       "https://ваш-сайт.kz";

    */

});
